# Our debug / test suite (docs/DEBUG_SUITE_PLAN.md): svr2011_dbg, never in a release. Only in its own build folder

# tools/release_guard.ps1 (package.ps1) fails a package holding it.
option(SVR2011_DEBUG_SUITE "Build the debug / test suite (svr2011_dbg) - never a release" OFF)
if(SVR2011_DEBUG_SUITE)
    get_filename_component(SVR2011_BUILD_NAME "${CMAKE_BINARY_DIR}" NAME)
    if(NOT SVR2011_BUILD_NAME MATCHES "^DebugSuite")
        message(FATAL_ERROR "SVR2011_DEBUG_SUITE is only for out/build/DebugSuite* (tools/dev/build_dbg.cmd), not ${CMAKE_BINARY_DIR}")
    endif()
    file(GLOB SVR2011_DEBUG_SOURCES CONFIGURE_DEPENDS ${CMAKE_CURRENT_SOURCE_DIR}/src/debug_suite/*.cpp)
    list(APPEND SVR2011_SOURCES ${SVR2011_DEBUG_SOURCES})
endif()

# The version and commit, for the log's first lines (src/build_info.h): the
# release version from VERSION.txt; the commit from BUILD_COMMIT.txt (git
# archive fills it in: export-subst), else from git (a working tree).
file(STRINGS "${CMAKE_CURRENT_SOURCE_DIR}/VERSION.txt" SVR2011_PORT_VERSION LIMIT_COUNT 1)
file(STRINGS "${CMAKE_CURRENT_SOURCE_DIR}/BUILD_COMMIT.txt" SVR2011_BUILD_COMMIT LIMIT_COUNT 1)
if(NOT SVR2011_BUILD_COMMIT OR SVR2011_BUILD_COMMIT MATCHES "Format")
    execute_process(COMMAND git -C "${CMAKE_CURRENT_SOURCE_DIR}" log -1 "--format=%h %cs"
                    OUTPUT_VARIABLE SVR2011_BUILD_COMMIT OUTPUT_STRIP_TRAILING_WHITESPACE ERROR_QUIET)
    execute_process(COMMAND git -C "${CMAKE_CURRENT_SOURCE_DIR}" status --porcelain --untracked-files=no
                    OUTPUT_VARIABLE SVR2011_DIRTY OUTPUT_STRIP_TRAILING_WHITESPACE ERROR_QUIET)
    if(SVR2011_DIRTY)
        string(APPEND SVR2011_BUILD_COMMIT " + local changes")
    endif()
endif()
if(NOT SVR2011_BUILD_COMMIT)
    set(SVR2011_BUILD_COMMIT "unknown")
endif()
set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS
             "${CMAKE_CURRENT_SOURCE_DIR}/VERSION.txt" "${CMAKE_CURRENT_SOURCE_DIR}/BUILD_COMMIT.txt")

if(WIN32)
    # The release version (VERSION.txt) for the programs' version information
    # (src/svr2011.rc, launcher/launcher.rc): version_info.h, e.g. 2,1,3,0 / "2.1.3.0".
    file(STRINGS "${CMAKE_CURRENT_SOURCE_DIR}/VERSION.txt" SVR2011_RC_VERSION LIMIT_COUNT 1)
    string(STRIP "${SVR2011_RC_VERSION}" SVR2011_RC_VERSION)
    string(REGEX MATCHALL "[0-9]+" SVR2011_RC_PARTS "${SVR2011_RC_VERSION}")
    list(LENGTH SVR2011_RC_PARTS SVR2011_RC_COUNT)
    while(SVR2011_RC_COUNT LESS 4)
        list(APPEND SVR2011_RC_PARTS 0)
        math(EXPR SVR2011_RC_COUNT "${SVR2011_RC_COUNT} + 1")
    endwhile()
    list(SUBLIST SVR2011_RC_PARTS 0 4 SVR2011_RC_PARTS)
    string(REPLACE ";" "," SVR2011_RC_NUM "${SVR2011_RC_PARTS}")
    string(REPLACE ";" "." SVR2011_RC_STR "${SVR2011_RC_PARTS}")
    file(CONFIGURE OUTPUT "${CMAKE_CURRENT_BINARY_DIR}/version_inc/version_info.h"
        CONTENT "#define SVR2011_VER_NUM ${SVR2011_RC_NUM}
#define SVR2011_VER_STR \"${SVR2011_RC_STR}\"
")
    set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS "${CMAKE_CURRENT_SOURCE_DIR}/VERSION.txt")
    # (on the resource scripts only: a new version doesn't rebuild the sources)
    set_property(SOURCE src/svr2011.rc launcher/launcher.rc APPEND PROPERTY INCLUDE_DIRECTORIES
        ${CMAKE_CURRENT_BINARY_DIR}/version_inc)
    # (src/utf8.manifest: UTF-8 code page - game folders with any letters)
    add_executable(svr2011 WIN32 ${SVR2011_SOURCES} src/svr2011.rc src/utf8.manifest)
elseif(ANDROID)
    # Android: libmain.so, which SDL's Java activity loads and starts (SDL_main).
    add_library(svr2011 SHARED ${SVR2011_SOURCES})
    set_target_properties(svr2011 PROPERTIES OUTPUT_NAME main)
    # Calls between the library's own functions (the recompiled game calls
    # its functions millions of times a second) bound directly, not through
    # the PLT (~3% of a phone's render thread in heavy entrances).
    target_compile_options(svr2011 PRIVATE -fno-semantic-interposition)
    target_link_options(svr2011 PRIVATE -Wl,-Bsymbolic-functions)
    # Custom GPU drivers (Mesa Turnip, ...): libadrenotools (BSD-2-Clause,
    # recomp/libadrenotools) opens libvulkan with the chosen driver; its hook
    # libraries go beside libmain.so in the APK (tools/build_apk.py).
    set(ADRENOTOOLS_DIR ${CMAKE_CURRENT_SOURCE_DIR}/../recomp/libadrenotools)
    if(EXISTS ${ADRENOTOOLS_DIR}/CMakeLists.txt)
        add_subdirectory(${ADRENOTOOLS_DIR} adrenotools EXCLUDE_FROM_ALL)
        target_sources(svr2011 PRIVATE src/gpu_driver.cpp)
        target_link_libraries(svr2011 PRIVATE adrenotools)
        target_compile_definitions(svr2011 PRIVATE SVR2011_ADRENOTOOLS=1)
        add_dependencies(svr2011 main_hook file_redirect_hook gsl_alloc_hook hook_impl)
    endif()
    # User entrance music: the phone's decoders (NDK media) into SDL audio.
    target_sources(svr2011 PRIVATE src/music.cpp)
    target_link_libraries(svr2011 PRIVATE mediandk)
else()
    add_executable(svr2011 ${SVR2011_SOURCES})
endif()

# DLSS (experimental, NVIDIA RTX GPUs, Direct3D 12): the NVIDIA DLSS SDK from OUTSIDE the repository (NVIDIA RTX SDK
# licence: never committed) - SVR2011_DLSS_SDK, else the DLSS_SDK environment variable, else
# ../../_sdk/dlss-310.9.16 beside the repository. Without it the game builds without DLSS. nvngx_dlss.dll is
# copied beside the exe for tests; the release zip gets it from tools/package.ps1.
if(SVR2011_FSR2)
    set(_svr2011_dlss_default "$ENV{DLSS_SDK}")
    if(NOT _svr2011_dlss_default)
        get_filename_component(_svr2011_dlss_default "${CMAKE_CURRENT_SOURCE_DIR}/../../_sdk/dlss-310.9.16" ABSOLUTE)
    endif()
    set(SVR2011_DLSS_SDK "${_svr2011_dlss_default}" CACHE PATH "NVIDIA DLSS SDK folder (include/, lib/) - optional")
    set(_svr2011_ngx_lib "${SVR2011_DLSS_SDK}/lib/Windows_x86_64/x64/nvsdk_ngx_d.lib")
    set(_svr2011_ngx_dll "${SVR2011_DLSS_SDK}/lib/Windows_x86_64/rel/nvngx_dlss.dll")
    if(EXISTS "${SVR2011_DLSS_SDK}/include/nvsdk_ngx.h" AND EXISTS "${_svr2011_ngx_lib}")
        message(STATUS "DLSS: SDK at ${SVR2011_DLSS_SDK}")
        target_sources(svr2011 PRIVATE src/native/dlss.cpp)
        target_include_directories(svr2011 PRIVATE "${SVR2011_DLSS_SDK}/include")
        target_link_libraries(svr2011 PRIVATE "${_svr2011_ngx_lib}")
        target_compile_definitions(svr2011 PRIVATE SVR2011_DLSS=1)
        if(EXISTS "${_svr2011_ngx_dll}")
            add_custom_command(TARGET svr2011 POST_BUILD
                COMMAND ${CMAKE_COMMAND} -E copy_if_different "${_svr2011_ngx_dll}" $<TARGET_FILE_DIR:svr2011>
                VERBATIM)
        endif()
    else()
        message(STATUS "DLSS: no SDK at ${SVR2011_DLSS_SDK} - built without DLSS")
    endif()
endif()

    endif()
endif()

rexglue_setup_target(svr2011 GPU_PLUGINS xenos)
# The SDK stages the GPU plugin (loaded at run time, never linked) only when
# the exe relinks: a change to the plugin alone would leave a stale copy
# beside the exe (and in Game Files / the release). Stage it on every build.
if(TARGET rexgpu-xenos AND NOT ANDROID)
    add_custom_target(svr2011_stage_gpu_plugin ALL
        COMMAND ${CMAKE_COMMAND} -E copy_if_different $<TARGET_FILE:rexgpu-xenos> $<TARGET_FILE_DIR:svr2011>
        DEPENDS rexgpu-xenos svr2011
        VERBATIM)
endif()
# FFmpeg and libmspack (LGPL) are DLLs of their own on Windows, beside the programs (Third-party notices.txt). The
# runtime with them: the exe doesn't relink when only the inside of the runtime changed, which left a stale copy here.
if(WIN32 AND TARGET rexffmpeg AND TARGET mspack)
    add_custom_target(svr2011_stage_lgpl_dlls ALL
        COMMAND ${CMAKE_COMMAND} -E copy_if_different $<TARGET_FILE:rexruntime> $<TARGET_FILE:rexffmpeg> $<TARGET_FILE:mspack> $<TARGET_FILE_DIR:svr2011>
        DEPENDS rexruntime rexffmpeg mspack svr2011
        VERBATIM)
endif()

    add_executable(svr2011_launcher WIN32 launcher/svr2011_launcher.c launcher/move_warning.c launcher/apk_package.c launcher/switch_package.c launcher/report.c launcher/report_send.c launcher/update_log.c launcher/report_receipts.c launcher/reports_tab.c launcher/suggestions_tab.c launcher/entrance_clash.c launcher/report_match.c launcher/report_flow.c launcher/stfs.c launcher/caw_import.c launcher/save_cleaner.c launcher/verify.c launcher/movie_maker.c launcher/bink_encode.c launcher/bink_decode.c launcher/updater.c launcher/release_check.c launcher/unzip.c launcher/mods_tab.c launcher/backgrounds.c launcher/music_tab.c launcher/roster_tab.c launcher/texpacks_tab.c launcher/ui_theme.c launcher/setup_tab.c launcher/discover_tab.c launcher/music_level.c launcher/nat_check.c launcher/launcher.rc src/utf8.manifest)
    set_target_properties(svr2011_launcher PROPERTIES OUTPUT_NAME "SvR2011 Launcher")
    # The hybrid CRT: the VC++ part (vcruntime) built in, the Universal CRT from Windows (ucrtbase.dll /
    # api-ms-win-crt-*, part of Windows 10+; Windows 7/8.1 with update KB2999226). The launcher starts
    # without the VC++ runtime, tells a player who has none / an old one where to get it (setup_tab.c), and
    # never holds the DLLs it may remove - and is an ordinary-sized program again (the fully static CRT
    # made it look packed to antivirus heuristics).
    set_property(TARGET svr2011_launcher PROPERTY MSVC_RUNTIME_LIBRARY "MultiThreaded")
    target_link_options(svr2011_launcher PRIVATE -Wl,/NODEFAULTLIB:libucrt.lib -Wl,/DEFAULTLIB:ucrt.lib)
    target_compile_definitions(svr2011_launcher PRIVATE
        UNICODE _UNICODE _CRT_SECURE_NO_WARNINGS WIN32_LEAN_AND_MEAN)
    target_link_options(svr2011_launcher PRIVATE -Wl,/ENTRY:wWinMainCRTStartup)

        comctl32 comdlg32 shell32 ole32 oleaut32 ws2_32 uxtheme windowscodecs mfplat mfreadwrite mfuuid propsys winhttp gdiplus delayimp
        bcrypt version rstrtmgr)
    # (Media Foundation only when used: Windows "N" without the Media Feature Pack)
    target_link_options(svr2011_launcher PRIVATE -Wl,/DELAYLOAD:mfplat.dll -Wl,/DELAYLOAD:mfreadwrite.dll)

    # launcher's updater compares it with the newest GitHub release).
    file(STRINGS "${CMAKE_CURRENT_SOURCE_DIR}/VERSION.txt" SVR2011_VERSION LIMIT_COUNT 1)
    target_compile_definitions(svr2011_launcher PRIVATE "PORT_VERSION=L\"${SVR2011_VERSION}\"")

    add_executable(svr2011_updater WIN32 launcher/updater_main.c launcher/updater.c launcher/update_log.c launcher/release_check.c launcher/unzip.c
        launcher/launcher.rc src/utf8.manifest)
    set_target_properties(svr2011_updater PROPERTIES OUTPUT_NAME "SvR2011 Updater")
    set_property(TARGET svr2011_updater PROPERTY MSVC_RUNTIME_LIBRARY "MultiThreaded")
    target_link_options(svr2011_updater PRIVATE -Wl,/NODEFAULTLIB:libucrt.lib -Wl,/DEFAULTLIB:ucrt.lib
        -Wl,/ENTRY:wWinMainCRTStartup)
    target_compile_definitions(svr2011_updater PRIVATE UNICODE _UNICODE _CRT_SECURE_NO_WARNINGS WIN32_LEAN_AND_MEAN
        SVR2011_RC_UPDATER "PORT_VERSION=L\"${SVR2011_VERSION}\"")
    target_link_libraries(svr2011_updater PRIVATE comctl32 shell32 winhttp bcrypt)
    add_dependencies(svr2011_launcher svr2011_updater)
